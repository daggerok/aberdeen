name: Update abrdn ETF data

on:
  schedule:
    - cron: '0 0 * * 0'
  workflow_dispatch:
    inputs:
      max_fetches:
        description: "Batch size; positive resumes cursor, 0 refreshes the selected universe and resets cursor"
        required: false
        default: "0"
        type: string
      request_sleep:
        description: "Seconds between request starts including retries; conservative shared gate"
        required: false
        default: "1"
        type: string
      concurrency:
        description: "Parallel fund workers; request starts remain conservatively paced"
        required: false
        default: "2"
        type: string
      aum:
        description: "Net assets min:max in USD; K/M/B/T or nano/micro/small/mid/large preset"
        required: false
        default: ":"
        type: string
      ter:
        description: "Gross expense ratio percent min:max"
        required: false
        default: ":"
        type: string
      dividend_yield:
        description: "Indicated dividend yield percent min:max"
        required: false
        default: ":"
        type: string
      sec_yield:
        description: "30-day SEC yield percent min:max"
        required: false
        default: ":"
        type: string
      tickers:
        description: "Ticker allowlist separated by spaces, commas or semicolons; empty means all"
        required: false
        default: ""
        type: string
      holdings_page_size:
        description: "Rows per holdings JSON page"
        required: false
        default: "250"
        type: string
      history_page_size:
        description: "Rows per market-price history JSON page"
        required: false
        default: "1000"
        type: string
      max_retries:
        description: "Retries after the first request; 0 disables retries"
        required: false
        default: "2"
        type: string
      history_range:
        description: "Yahoo daily history range: max or Ny (e.g. 5y); preserves prior history"
        required: false
        default: "max"
        type: string
      store_raw_downloads:
        description: "Save source JSON under api/aberdeen/raw"
        required: false
        default: "false"
        type: string
      sec_ua:
        description: "SEC User-Agent; supply a real identifying contact for automated access"
        required: false
        default: "daggerok Aberdeen ETF feed (https://github.com/daggerok/aberdeen)"
        type: string
      skip_yahoo:
        description: "Skip Yahoo history and dividends; retain published data"
        required: false
        default: "false"
        type: string
      skip_aberdeen:
        description: "Use published catalog/details; only fallback providers are called"
        required: false
        default: "false"
        type: string
      edgar_fallback:
        description: "Enable SEC N-PORT holdings fallback; never use an unrelated series"
        required: false
        default: "true"
        type: string
      verbose:
        description: "Print per-request failures and fallback diagnostics"
        required: false
        default: "false"
        type: string
      performance_ytd:
        description: "Annualized YTD return percent min:max"
        required: false
        default: ":"
        type: string
      performance_1y:
        description: "Annualized 1Y return percent min:max"
        required: false
        default: ":"
        type: string
      performance_3y:
        description: "Annualized 3Y return percent min:max"
        required: false
        default: ":"
        type: string
      performance_5y:
        description: "Annualized 5Y return percent min:max"
        required: false
        default: ":"
        type: string
      performance_10y:
        description: "Annualized 10Y return percent min:max"
        required: false
        default: ":"
        type: string
      total_return_ytd:
        description: "Cumulative YTD return percent min:max"
        required: false
        default: ":"
        type: string
      total_return_1y:
        description: "Cumulative 1Y return percent min:max"
        required: false
        default: ":"
        type: string
      total_return_3y:
        description: "Cumulative 3Y return percent min:max"
        required: false
        default: ":"
        type: string
      total_return_5y:
        description: "Cumulative 5Y return percent min:max"
        required: false
        default: ":"
        type: string
      total_return_10y:
        description: "Cumulative 10Y return percent min:max"
        required: false
        default: ":"
        type: string

permissions:
  contents: write

concurrency:
  group: update-data
  cancel-in-progress: false

jobs:
  update-data:
    runs-on: ubuntu-latest
    env:
      MAX_FETCHES: ${{ inputs.max_fetches || '0' }}
      REQUEST_SLEEP: ${{ inputs.request_sleep || '1' }}
      CONCURRENCY: ${{ inputs.concurrency || '2' }}
      AUM: ${{ inputs.aum || ':' }}
      TER: ${{ inputs.ter || ':' }}
      DIVIDEND_YIELD: ${{ inputs.dividend_yield || ':' }}
      SEC_YIELD: ${{ inputs.sec_yield || ':' }}
      TICKERS: ${{ inputs.tickers || '' }}
      HOLDINGS_PAGE_SIZE: ${{ inputs.holdings_page_size || '250' }}
      HISTORY_PAGE_SIZE: ${{ inputs.history_page_size || '1000' }}
      MAX_RETRIES: ${{ inputs.max_retries || '2' }}
      HISTORY_RANGE: ${{ inputs.history_range || 'max' }}
      STORE_RAW_DOWNLOADS: ${{ inputs.store_raw_downloads || 'false' }}
      SEC_UA: ${{ inputs.sec_ua || 'daggerok Aberdeen ETF feed (https://github.com/daggerok/aberdeen)' }}
      SKIP_YAHOO: ${{ inputs.skip_yahoo || 'false' }}
      SKIP_ABERDEEN: ${{ inputs.skip_aberdeen || 'false' }}
      EDGAR_FALLBACK: ${{ inputs.edgar_fallback || 'true' }}
      VERBOSE: ${{ inputs.verbose || 'false' }}
      PERFORMANCE_YTD: ${{ inputs.performance_ytd || ':' }}
      PERFORMANCE_1Y: ${{ inputs.performance_1y || ':' }}
      PERFORMANCE_3Y: ${{ inputs.performance_3y || ':' }}
      PERFORMANCE_5Y: ${{ inputs.performance_5y || ':' }}
      PERFORMANCE_10Y: ${{ inputs.performance_10y || ':' }}
      TOTAL_RETURN_YTD: ${{ inputs.total_return_ytd || ':' }}
      TOTAL_RETURN_1Y: ${{ inputs.total_return_1y || ':' }}
      TOTAL_RETURN_3Y: ${{ inputs.total_return_3y || ':' }}
      TOTAL_RETURN_5Y: ${{ inputs.total_return_5y || ':' }}
      TOTAL_RETURN_10Y: ${{ inputs.total_return_10y || ':' }}

    steps:
      - name: Checkout
        uses: actions/checkout@v7
      - name: Setup bun
        uses: oven-sh/setup-bun@v2
        with:
          bun-version: latest
      - name: Install dependencies
        run: bun install --frozen-lockfile
      - name: Run updater unit tests
        run: bun test scripts/update-data.test.ts
      - name: Generate api/aberdeen static data
        run: bun ./scripts/update-data.ts
      - name: Commit updated data
        if: ${{ !cancelled() }}
        run: |
          git add api/aberdeen
          if git diff --cached --quiet -- api/aberdeen; then
            echo "api/aberdeen is already up to date"
            exit 0
          fi
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git commit -m "chore(data): update abrdn ETF data"
          git push
